cmake_minimum_required(VERSION 3.8)
project(omnifleet_dwb_hot)
find_package(ament_cmake REQUIRED)
find_package(dwb_core REQUIRED)
find_package(nav2_core REQUIRED)
find_package(pluginlib REQUIRED)
add_library(${PROJECT_NAME} SHARED src/hot_dwb.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME} dwb_core nav2_core pluginlib)
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
install(TARGETS ${PROJECT_NAME} EXPORT export_${PROJECT_NAME} LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(dwb_critics REQUIRED)
  ament_add_gtest(test_hot_rotation test/test_hot_rotation.cpp)
  target_link_libraries(test_hot_rotation ${PROJECT_NAME})
  ament_target_dependencies(test_hot_rotation dwb_core dwb_critics)
  ament_add_gtest(test_hot_scoring test/test_hot_scoring.cpp)
  target_link_libraries(test_hot_scoring ${PROJECT_NAME})
  ament_target_dependencies(test_hot_scoring dwb_core)
endif()
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_dependencies(dwb_core nav2_core pluginlib)
ament_package()
