cmake_minimum_required(VERSION 3.8)
project(omnifleet_t2_mola_experiments)
find_package(ament_cmake REQUIRED)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/fresh_pointcloud_filter.py scripts/base_link_rslidar_tf.py scripts/foxglove_layer_offset.py scripts/cleanup_unified_application_nodes.py DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/nav2_readiness_gate.py DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/lidar_velocity.py scripts/mola_realtime_nav_grid.py DESTINATION lib/${PROJECT_NAME})
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(omnifleet_localization REQUIRED)
add_executable(nav_cloud_deskew src/nav_cloud_deskew.cpp)
target_compile_features(nav_cloud_deskew PRIVATE cxx_std_17)
target_compile_options(nav_cloud_deskew PRIVATE -O3)
target_link_libraries(nav_cloud_deskew Eigen3::Eigen)
ament_target_dependencies(nav_cloud_deskew rclcpp sensor_msgs nav_msgs geometry_msgs std_msgs tf2_ros)
install(TARGETS nav_cloud_deskew DESTINATION lib/${PROJECT_NAME})
add_executable(airy_imu_adapter_buffered src/airy_imu_adapter_buffered.cpp)
target_compile_features(airy_imu_adapter_buffered PRIVATE cxx_std_17)
ament_target_dependencies(airy_imu_adapter_buffered rclcpp sensor_msgs omnifleet_localization)
install(TARGETS airy_imu_adapter_buffered DESTINATION lib/${PROJECT_NAME})
ament_package()
