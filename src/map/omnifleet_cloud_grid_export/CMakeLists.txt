cmake_minimum_required(VERSION 3.8)
project(omnifleet_cloud_grid_export)
find_package(ament_cmake REQUIRED)
find_package(PCL REQUIRED COMPONENTS io)
add_executable(pcd_to_xyz src/pcd_to_xyz.cpp)
target_compile_features(pcd_to_xyz PRIVATE cxx_std_17)
target_include_directories(pcd_to_xyz PRIVATE ${PCL_INCLUDE_DIRS})
target_link_libraries(pcd_to_xyz ${PCL_LIBRARIES})
install(TARGETS pcd_to_xyz DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/cloud_to_grid DESTINATION lib/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_package()
