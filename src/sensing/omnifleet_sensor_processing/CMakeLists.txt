cmake_minimum_required(VERSION 3.8)
project(omnifleet_sensor_processing)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(Eigen3 REQUIRED)
ament_python_install_package(${PROJECT_NAME})
add_library(omnifleet_sensor_conversions src/conversions.cpp)
target_compile_features(omnifleet_sensor_conversions PUBLIC cxx_std_17)
target_include_directories(omnifleet_sensor_conversions PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
ament_target_dependencies(omnifleet_sensor_conversions sensor_msgs)
foreach(node airy_imu_adapter_node airy_imu_adapter_buffered)
  add_executable(${node} src/${node}.cpp)
  target_compile_features(${node} PRIVATE cxx_std_17)
  target_link_libraries(${node} omnifleet_sensor_conversions)
  ament_target_dependencies(${node} rclcpp sensor_msgs)
  install(TARGETS ${node} DESTINATION lib/${PROJECT_NAME})
endforeach()
add_executable(nav_cloud_deskew src/nav_cloud_deskew.cpp)
target_compile_features(nav_cloud_deskew PRIVATE cxx_std_17)
target_compile_options(nav_cloud_deskew PRIVATE -O3)
target_link_libraries(nav_cloud_deskew Eigen3::Eigen)
ament_target_dependencies(nav_cloud_deskew rclcpp sensor_msgs nav_msgs geometry_msgs std_msgs tf2_ros)
install(TARGETS nav_cloud_deskew DESTINATION lib/${PROJECT_NAME})
install(TARGETS omnifleet_sensor_conversions EXPORT export_${PROJECT_NAME}
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_include_directories(include)
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_libraries(omnifleet_sensor_conversions)
ament_export_dependencies(sensor_msgs)
install(PROGRAMS scripts/usb_camera_publisher scripts/pointcloud_radius_filter
  scripts/fresh_pointcloud_filter.py scripts/lidar_velocity.py scripts/sensor_clock_guard.py
  DESTINATION lib/${PROJECT_NAME})
ament_package()
